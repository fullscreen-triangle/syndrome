\chapter{Poincaré Computing}
\label{ch:poincare}

Forward simulation of biochemical networks has been the computational workhorse
of systems biology for three decades.  This chapter argues that it is the wrong
tool.  The forward simulation problem is intractable for realistic cellular
networks; the intractability is not a technical limitation awaiting faster
hardware but a structural consequence of asking the wrong question.  The right
question, motivated by Poincaré's recurrence theorem, is a \emph{trajectory
completion} problem: given sparse observations, recover the most probable full
trajectory.  We show this problem is solvable in time $\mathcal{O}(k)$ where
$k$ is the partition depth, versus $\mathcal{O}(e^{kT})$ for forward simulation.

%% ---------------------------------------------------------------------------
\section{The Forward Simulation Paradox}
\label{sec:forward-paradox}
%% ---------------------------------------------------------------------------

Consider a biochemical network with $k$ species and $r$ reactions, governed by
the ODE
\begin{equation}
    \dot x \;=\; S\,v(x),
    \label{eq:ode}
\end{equation}
where $S\in\mathbb{Z}^{k\times r}$ is the stoichiometric matrix and
$v:\mathbb{R}^k_+\to\mathbb{R}^r_+$ the rate vector.

\begin{theorem}[Forward Simulation Intractability]
\label{thm:intractability}
Numerically integrating \Cref{eq:ode} over a time interval $[0,T]$ for a
biochemical network with $k \sim 10^4$ species (typical metabolome) requires
$\mathcal{O}(e^{kT})$ operations in the worst case, due to stiffness ratios
that scale exponentially with $k$.
\end{theorem}

\begin{remark}
The exponential blowup originates from the separation of timescales: fast
reactions (electron transfer, $\sim 10^{-15}$~s) and slow reactions (gene
expression, $\sim 10^3$~s) coexist in the same system.  An explicit integrator
must step at the fastest timescale; the number of steps is $T/\tau_\mathrm{fast}
\approx 10^{18}$ for one cell cycle.  Implicit methods reduce step count but
require solution of $k\times k$ nonlinear systems at each step.
\end{remark}

\begin{proposition}[Poincaré Recurrence Constraint]
\label{prop:poincare}
Biochemical trajectories are not free trajectories in $\mathbb{R}^k_+$.  By
the Poincaré recurrence theorem applied to the bounded phase space
$\Omega_\mathrm{cell}$ (\Cref{def:colloid}), every trajectory returns
arbitrarily close to its initial condition in finite time.  The accessible
phase space volume is a tiny fraction of $\mathbb{R}^k_+$; trajectories are
constrained by conservation laws, partition depth gradients, and aperture
conditions.
\end{proposition}

The practical consequence of \Cref{prop:poincare} is decisive: instead of
integrating forward through the vast ambient space, one should
\emph{complete} the trajectory on the constrained recurrent set.

\section{Trit-Trajectory Correspondence}
\label{sec:trit-traj}
%% ---------------------------------------------------------------------------

\begin{definition}[Trit Representation]
\label{def:trit}
Represent the partition state of the biochemical network as a ternary string
$\mathbf{t} \in \{-1,0,+1\}^n$, where each trit $t_i$ encodes whether the
$i$-th partition coordinate is:
\[
    t_i = \begin{cases}
        -1 & \text{system is below the $i$-th partition boundary,} \\
        \phantom{-}0 & \text{system is at the $i$-th boundary (transitioning),} \\
        +1 & \text{system is above the $i$-th boundary.}
    \end{cases}
\]
\end{definition}

\begin{definition}[Trit Trajectory]
\label{def:trit-traj}
A \emph{trit trajectory} is a finite sequence $\mathbf{t}^{(0)},
\mathbf{t}^{(1)}, \ldots, \mathbf{t}^{(T)}$ of trit strings, where each
consecutive pair differs by exactly one trit flip (a single partition boundary
crossing).
\end{definition}

\begin{theorem}[Trit-Trajectory Correspondence]
\label{thm:trit-traj}
Every biochemical trajectory $\Traj:[0,T]\to\Omega_\mathrm{cell}$ obeying
\Cref{eq:ode} projects, under $\Pi_n$, to a valid trit trajectory.  Conversely,
every valid trit trajectory (one satisfying the Kirchhoff and thermodynamic
constraints of \Cref{sec:circuit-dynamics}) lifts to a family of biochemical
trajectories, unique up to within-cell degeneracy.
\end{theorem}

The trit representation collapses the $k$-dimensional continuous trajectory to
an $n$-dimensional discrete one, where $n$ is the partition depth.  This
dimensional reduction is the key computational gain.

\section{The Trajectory Completion Problem}
\label{sec:completion-problem}
%% ---------------------------------------------------------------------------

  \label{fig:ch05-partition-geometry}

\begin{definition}[Trajectory Completion Problem]
\label{def:completion}
Given partial observations $\{x_{t_1}, x_{t_2}, \ldots, x_{t_m}\}$ with
$m \ll T$, find the maximum a posteriori (MAP) trajectory:
\begin{equation}
    \Traj^* \;=\; \arg\max_{\Traj \text{ valid}} P(\Traj \mid x_{t_1},\ldots,x_{t_m}),
    \label{eq:map-traj}
\end{equation}
where ``valid'' means $\Traj$ satisfies the Kirchhoff (\Cref{thm:kcl}),
thermodynamic (\Cref{thm:kvl}), and aperture constraints
(\Cref{def:aperture}).
\end{definition}

\begin{theorem}[Backward Propagation Complexity]
\label{thm:backward-complexity}
Given an observation $x$ at time $t$, backward propagation through the
categorical constraints to determine all consistent predecessors has complexity
$\mathcal{O}(k)$ where $k$ is the partition depth (number of partition levels).
This is exponentially faster than forward simulation: $\mathcal{O}(k)$
versus $\mathcal{O}(e^{kT})$.
\end{theorem}

\begin{proof}
At each partition level $\ell = k, k-1, \ldots, 1$, backward propagation
eliminates those predecessor states inconsistent with the observation.  The
elimination at level $\ell$ takes $\mathcal{O}(|\Partition_\ell|)$ time, which
is $\mathcal{O}(1)$ per level under the hierarchical partition structure
(each cell has at most $\Capacity(n) = 2n^2$ sub-cells).  Summing over $k$
levels gives $\mathcal{O}(k)$ total.
\end{proof}

\begin{definition}[MAP Backward Trajectory via Viterbi]
\label{def:viterbi}
The MAP trit trajectory is computed by the Viterbi algorithm on the trit string.
At each trit position $i$, the optimal flip is the one maximising the posterior
under the joint constraint:
\begin{enumerate}[label=(\alph*)]
    \item \textbf{Kirchhoff constraint}: the resulting trit string obeys
          current balance at every node (\Cref{thm:kcl});
    \item \textbf{Thermodynamic constraint}: the flip is along the partition
          depth gradient (\Cref{thm:partition-flux});
    \item \textbf{Observation constraint}: the trajectory passes through the
          observed cells $\{P_n(x_{t_j})\}$.
\end{enumerate}
\end{definition}

\begin{theorem}[Time-Invariance of MAP Backward Trajectory]
\label{thm:time-invariance}
For autonomous kinetics (no explicit time dependence in $v(x)$), the MAP
backward trajectory of any network node is \emph{time-invariant}: it depends
only on the current partition state $\mathbf{t}$ and the network topology
$\Gcond$, not on absolute time $t$.  The MAP backward trajectory is therefore
a \emph{categorical address} --- a fixed geometric identity of the node in
partition space.
\end{theorem}

\begin{proof}
Autonomous kinetics means $v(x,t) = v(x)$.  The Viterbi recursion at each
partition level depends on transition rates $G_{ij}(x)$ and the observation
residuals.  Since $G_{ij}$ depends on $x$ (via \Cref{def:edge-cond}) but not
on $t$, and the observation constraints are formulated in terms of partition
cells (not absolute times), the Viterbi output is a function of $\mathbf{t}$
and $\Gcond$ alone.
\end{proof}

\section{Completion Algorithm}
\label{sec:completion-alg}
%% ---------------------------------------------------------------------------

\end{figure}

\begin{definition}[Project-Complete-Compose Algorithm]
\label{def:pcc}
The \emph{Project-Complete-Compose} (PCC) algorithm for trajectory completion
proceeds in three steps:
\begin{enumerate}[label=\textbf{Step \arabic*:}, leftmargin=*, itemsep=4pt]
    \item \textbf{Project.}  Map each partial observation $x_{t_j}$ to its
          partition cell: $c_j = \Pi_n(x_{t_j}) \in \Partition_n$.

    \item \textbf{Complete.}  Find the MAP backward trit trajectory
          $\mathbf{t}^* = \{t^{*(0)}, \ldots, t^{*(T)}\}$ connecting the
          projected cells $\{c_j\}$ under the constraints of
          \Cref{def:viterbi}, using the Viterbi algorithm.

    \item \textbf{Compose.}  Compose the backward maps
          $\{\Back_\ell\}_{\ell=1}^k$ along $\mathbf{t}^*$ to recover the full
          partition-state sequence, and lift back to a representative
          concentration trajectory $\{x_t^*\}$ via the inverse
          partition map.
\end{enumerate}
\end{definition}

\begin{theorem}[PCC Convergence]
\label{thm:pcc-convergence}
The PCC algorithm converges to a fixed point of the trajectory completion map
on the space of fuzzy state tuples equipped with the Hausdorff metric.  The
contraction factor is
\begin{equation}
    \lambda
    \;=\;
    \max_{i,j}\frac{|G_{ij}|}{\sum_j |G_{ij}|}
    \;<\; 1,
    \label{eq:contraction}
\end{equation}
so convergence follows from the Banach fixed-point theorem.
\end{theorem}

\begin{proof}
The completion map $\mathcal{T}$ sends a fuzzy trajectory to the MAP backward
trajectory consistent with it.  Under the Hausdorff metric $d_H$ on fuzzy state
tuples, $\mathcal{T}$ is a contraction with factor $\lambda$ given by the
normalised maximum off-diagonal conductance.  Since $G_{ij} \ge 0$ and
$G_{ij} < \sum_j G_{ij}$ (each row has a non-trivial diagonal), $\lambda < 1$.
By Banach's theorem, $\mathcal{T}$ has a unique fixed point to which iteration
converges geometrically.
\end{proof}

\begin{remark}[Complexity Summary]
PCC has total complexity $\mathcal{O}(m \cdot k)$ for $m$ observations and
partition depth $k$.  Forward simulation has complexity $\mathcal{O}(e^{kT})$.
For the metabolome ($k \sim 10^4$, $m \sim 10^2$, $T \sim 10^3$~s), the ratio
exceeds $10^{9990}$, which is not a practical but a structural impossibility for
forward simulation.
\end{remark}

\section{Categorical Apertures as Constraints}
\label{sec:aperture-constraints}
%% ---------------------------------------------------------------------------

The aperture picture of \Cref{sec:protein-function} and \Cref{sec:aperture}
enters the completion algorithm as a dimensionality-reduction device.

\begin{theorem}[Aperture Dimensionality Reduction]
\label{thm:aperture-dim}
Each enzyme $E$ with active-site cell $\mathcal{C}_E$ eliminates all trajectory
branches not passing through $\mathcal{C}_E$.  If the network contains $N_E$
distinct enzymes with non-overlapping aperture cells, the effective search
dimensionality for trajectory completion is reduced from $n$ (total partition
depth) to $n - N_E$ (residual degrees of freedom).
\end{theorem}

\begin{proof}
Trajectories must pass through every aperture cell dictated by the reaction
network structure.  Each aperture cell $\mathcal{C}_E$ pins one partition
coordinate: the trajectory is required to have $\Pi_{n_E}(x_t) = \mathcal{C}_E$
at some time $t$.  If $N_E$ non-overlapping cells are pinned, $N_E$ coordinates
are fixed, reducing effective dimensionality by $N_E$.
\end{proof}

\begin{definition}[Aperture Traversal Order]
\label{def:traversal-order}
In the PCC algorithm, aperture constraints are applied in \emph{decreasing}
order of partition depth: the most specific apertures (largest $\Depth_E$) are
applied first.  This maximises early pruning and minimises the Viterbi search
tree.
\end{definition}

The traversal order of \Cref{def:traversal-order} is the categorical analogue
of branch-and-bound search: by fixing the most constrained variables first, the
algorithm eliminates the largest number of infeasible trajectories before
branching.

\begin{proposition}[Connection to Information Catalysis]
\label{prop:aperture-info}
The information-catalytic role of enzymes (\Cref{thm:info-catalysis}) and their
computational role as dimensionality-reducing constraints
(\Cref{thm:aperture-dim}) are two faces of the same phenomenon.  Reducing
$\dcat(S,P)$ (information-catalytic view) is equivalent to reducing the
effective search dimension for the trajectory (computational view): both
quantify the extent to which the enzyme resolves categorical ambiguity.
\end{proposition}

\section{Biochemical Validation}
\label{sec:poincare-validation}
%% ---------------------------------------------------------------------------

We validate the PCC framework against the fastest known enzyme, carbonic
anhydrase~II (CA-II), whose catalytic mechanism is well characterised and whose
rate constants span the widest dynamic range of any benchmarked enzyme.

\begin{example}[Carbonic Anhydrase II]
\label{ex:ca2}
CA-II catalyses $\mathrm{CO}_2 + \mathrm{H}_2\mathrm{O} \rightleftharpoons
\mathrm{HCO}_3^- + \mathrm{H}^+$ with $k_\mathrm{cat} = 10^6$~s$^{-1}$, the
highest known turnover number.

\begin{enumerate}[label=(\roman*), itemsep=4pt]
    \item \textbf{Forward simulation cost.}  Integrating the CA-II reaction
          mechanism (17 elementary steps, including proton relay through the
          Grotthuss network) at the timescale set by proton transfer
          ($\tau \sim 10^{-12}$~s) over one catalytic cycle
          ($T \sim 10^{-6}$~s) requires $\mathcal{O}(10^{12})$ steps.

    \item \textbf{Trajectory completion cost.}  The relevant partition depth
          for CA-II is $k = \log_2(k_\mathrm{cat}/k_\mathrm{basal}) \approx
          \log_2(10^6) \approx 20$.  PCC requires $\mathcal{O}(20)$ backward
          propagation steps to complete the trajectory from any single
          observation.

    \item \textbf{Speedup.}  The ratio of forward to completion cost is
          $10^{12}/20 = 5\times 10^{10}$: a factor exceeding $10^{10}$.
\end{enumerate}
\end{example}

\begin{remark}
The speedup in \Cref{ex:ca2} is not specific to CA-II.  Any enzyme with
$k_\mathrm{cat} \gg k_\mathrm{basal}$ exhibits a partition depth $k =
\log_2(k_\mathrm{cat}/k_\mathrm{basal})$ much smaller than the number of
integration steps required by forward simulation.  The $10^{10}$ factor is
therefore a lower bound for the class of enzymes operating near the diffusion
limit.
\end{remark}

\section{The Prototype Scripting Language}
\label{sec:scripting}
%% ---------------------------------------------------------------------------

The PCC framework suggests a language for specifying biochemical models
without differential equations.

\subsection{Design Principles}

The language obeys a single semantic rule: \emph{no laws, only constraints}.
Every statement specifies a partition constraint --- a set of cells in which the
current state must lie.  There are no rate laws, no differential equations, no
parameters.  The ``dynamics'' of the system are recovered by trajectory
completion on the constraint graph.

This is the same principle as the partition algebra of \Cref{ch:epistemic}: the
partition hierarchy defines which cells are accessible; the trajectory is the
sequence of cells visited, computed by PCC rather than by integration.

\subsection{Language Semantics}

\begin{definition}[Constraint Statement]
\label{def:constraint-stmt}
A \emph{constraint statement} is a set expression
$\{C_1, C_2, \ldots, C_r\}$ asserting that the current partition state
$\mathbf{t}$ must satisfy $\mathbf{t} \in \bigcup_{i=1}^r C_i$.  Each $C_i$
is a partition cell (a trit string or a logical combination of trit conditions).
\end{definition}

\begin{definition}[Program Execution]
\label{def:prog-exec}
A \emph{program} is a sequence of constraint statements
$\mathcal{S}_1, \mathcal{S}_2, \ldots, \mathcal{S}_p$.  Execution proceeds by:
\begin{enumerate}[label=(\arabic*)]
    \item Initialise the partition state $\mathbf{t}^{(0)}$ from the first
          available observation (or a prior).
    \item For each statement $\mathcal{S}_j$, restrict the admissible set of
          partition states to $\mathcal{S}_j$.
    \item Apply PCC to complete the trajectory through the intersection of all
          admissible sets.
    \item Return the MAP trajectory $\Traj^*$ as the program output.
\end{enumerate}
\end{definition}

\begin{proposition}[Equivalence to Partition Algebra]
\label{prop:lang-equivalence}
Program execution (\Cref{def:prog-exec}) is equivalent to applying the
partition composition operations of \Cref{sec:cellular-processes} in sequence.
Each constraint statement implements one partition read; the PCC step composes
them to obtain the full state.
\end{proposition}

\begin{example}[Constraint Program for Glycolytic Oscillations]
\label{ex:glycolysis}
The glycolytic oscillator (Selkov model) can be specified as the constraint
program:
\begin{align*}
    &\{C_\mathrm{ADP} > C_\mathrm{ATP}\}
    &&\text{(energy charge constraint)} \\
    &\{C_\mathrm{F6P} \in [\underline{c},\bar{c}]\}
    &&\text{(fructose-6-phosphate window)} \\
    &\{G_{ij} > G_\mathrm{threshold}\text{ for PFK edge}\}
    &&\text{(phosphofructokinase aperture active)} \\
    &\{\Hol_\ell = 0 \text{ for all cycles } \ell\}
    &&\text{(thermodynamic consistency)}
\end{align*}
Executing this program via PCC recovers the oscillatory trajectory without
specifying any rate constants or differential equations.
\end{example}

\begin{remark}[Biological Interpretation]
\Cref{ex:glycolysis} illustrates the conceptual shift: biological systems do
not obey rate laws; they inhabit constraint sets.  The rate laws of biochemistry
($v = V_\mathrm{max}[S]/(K_m+[S])$, etc.) are \emph{derived} quantities ---
the MAP trajectory through the constraint graph, not primitive axioms.  The
scripting language makes this derivation explicit and computable.
\end{remark}
